cmake_minimum_required(VERSION 3.16)
project(os_boot_demo C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(SDL2 REQUIRED)
add_executable(os-boot boot_demo.c)
if(TARGET SDL2::SDL2)
  target_link_libraries(os-boot PRIVATE SDL2::SDL2)
else()
  target_include_directories(os-boot PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(os-boot PRIVATE ${SDL2_LIBRARIES})
endif()
if(UNIX AND NOT APPLE)
  target_link_libraries(os-boot PRIVATE m)
endif()
